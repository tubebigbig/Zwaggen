- [ ] **Step 3: Add collision detector + apply sanitization in `types.ts`**

At the top of `generateTs`:

```ts
function detectKeyCollisions(spec: Spec): void {
  const seen = new Map<string, string>();
  for (const key of Object.keys(spec.types)) {
    const s = sanitizeFolderKey(key);
    const prior = seen.get(s);
    if (prior !== undefined && prior !== key) {
      throw new Error(`Codegen key collision: ${prior} and ${key} both sanitize to ${s}. Rename one.`);
    }
    seen.set(s, key);
  }
}

export function generateTs(spec: Spec): string {
  detectKeyCollisions(spec);
  const lines: string[] = [HEADER];
  for (const [name, def] of Object.entries(spec.types)) {
    lines.push(emitTopLevel(sanitizeFolderKey(name), def, spec));
    lines.push('');
  }
  return lines.join('\n');
}
```

```ts
function tsRefType(def: TypeDef, spec: Spec): string {
  switch (def.kind) {
    case 'string': return 'string';
    case 'number':
    case 'integer': return 'number';
    case 'boolean': return 'boolean';
    case 'null': return 'null';
    case 'literal': return typeof def.value === 'string' ? `'${def.value}'` : String(def.value);
    case 'array': return `${tsRefType(def.element, spec)}[]`;
    case 'union': return def.variants.map((v) => tsRefType(v, spec)).join(' | ');
    case 'ref': return sanitizeFolderKey(def.ref);
    case 'object': {
      if (def.fields.length === 0) return '{}';
      const fields = def.fields.map((f) => {
        const opt = f.required ? '' : '?';
        return `${safeIdentifier(f.name)}${opt}: ${tsRefType(f.type, spec)}`;
      }).join('; ');
      return `{ ${fields} }`;
    }
  }
}
```

- [ ] **Step 2: Implement `camelizeTag` in `helpers.ts`**

```ts
export function camelizeTag(tag: string): string {
  if (tag === 'default') return 'default';
  const parts = tag.split(/[^A-Za-z0-9]+/).filter(Boolean);
  if (parts.length === 0) return 'default';
  const [first, ...rest] = parts;
  const camelFirst = first!.charAt(0).toLowerCase() + first!.slice(1);
  const camelRest = rest.map((p) => p.charAt(0).toUpperCase() + p.slice(1));
  return [camelFirst, ...camelRest].join('');
}
```
